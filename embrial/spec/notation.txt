SYNTAX > Productions;

EOF         > '\0';
CONTROL     > EOF | '\u0001' | '\u0002' | ... | '\u0037';
NEWLINE     > '\n';

CHR_WHITESPACE  > '\n' | '\t' | ' ';
OR              > '|';
QUOTE           > '\'';
DOUBLE_QUOTE    > '"';

SYM             > '!' | DOUBLE_QUOTE | '#' | ... | '/' | ':' | ';' | '<' | ... | '@' | '[' | '\' | ']' | ... | QUOTE | '{' | OR | '}' | '~';
ALPHABETIC      > 'a' | 'b' | 'c' | ... | 'z' | 'A' | 'B' | 'C' | ... | 'Z';
NUMERIC         > '0' | '1' | '2' | ... | '9';

EPSILON >;
ID      > ALPHABETIC lettersOpt | '_';
    lettersOpt  > letters | EPSILON;
    letters     > idValid lettersOpt;
    idValid     > ALPHABETIC | NUMERIC | '_';
NUM     > NUMERIC numbersOpt;
    numbersOpt  > numbers | EPSILON;
    numbers     > numValid numbersOpt
    numValid    > NUMERIC | '_';

Productions > Production ProductionOpt;
    Production > NonTerminal rhs;
        NonTerminal > ID;
        rhs > '>' expressionOpt;
            expressionOpt   > expression | EPSILON;
            expression      > Terminal expression | NonTerminals;
                Terminal        > symbol | keyword | "EPSILON" | "EOF" | "ID" | "SYM" | "NUM";
                    symbol > QUOTE charachter QUOTE;
                    charachter > '\' escaped | unescaped;
                        escaped > '\''| '\\' | '\/' | 'b' | 'f' | 'n' | 'r' | 't' | 'u' NUMERIC NUMERIC NUMERIC NUMERIC
                        unescaped > SYM | ALPHABETIC | NUMERIC;

                NonTerminals    > ';' | NonTerminal NonTerminals;
    ProductionOpt > whitespace Productions | EOF;
        whitespace  > CHR_WHITESPACE whitespace | '#' commentBody whitespace | EPSILON;
        commentBody > commentable commentOpt;
            commentable > SYM | ALPHABETIC | NUMERC | '\t' | ' ';
            commentOpt > '\n' whitespace | commentBody
